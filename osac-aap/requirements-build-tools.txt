# prefetch them for the hermetic build; update these pins if
# execution-environment.yaml's ansible_core/ansible_runner versions change,
# or if ansible-builder's own template (base/builder/final stage pip
# installs) changes in a future ansible-builder upgrade.
ansible-core==2.20.3

# bindep's own transitive dependencies -- Cachi2 didn't auto-resolve these
# for a build-requirements entry the way it did for ansible-core/
# ansible-runner's own transitive deps (confirmed live: distro>=1.7 missing
# from the prefetch, even though bindep==2.14.0 itself was present), so
# list them explicitly. Resolved via `pip install bindep==2.14.0 --dry-run
# --report` against a clean environment.
distro==1.9.0
pbr==7.1.3
Parsley==1.3
# ansible-core ships no wheel, only an sdist, so pip must build it -- which
# needs its own build-system requirement (setuptools<=82.0.0,>=66.1.0, per
# ansible-core's own pyproject.toml) prefetched too. Without this explicit
# entry, Cachi2 only prefetches whatever version happened to resolve for
# some OTHER package's transitive dependency (82.0.1 here), which doesn't
# satisfy ansible-core's own upper bound -- confirmed live: "Could not find
# a version that satisfies the requirement setuptools<=82.0.0,>=66.1.0".
# ansible-runner's own build-system.requires is a much older, narrower
# range (setuptools<=70.0.0,>=45, confirmed live: "Could not find a
# version that satisfies the requirement setuptools<=70.0.0,>=45") --
# 70.0.0 is the single version satisfying both packages' constraints at
# once, so one pin covers both rather than needing two coexisting
# versions.
setuptools==70.0.0
# ansible-core's build-system.requires also hard-pins wheel == 0.45.1
# exactly (confirmed via its real sdist pyproject.toml), a different
# exact version than python-esiclient's own wheel need below -- both
# coexist fine in the shared prefetch cache since pip resolves each
# package's isolated build environment independently. Confirmed live:
# "Could not find a version that satisfies the requirement
# wheel==0.45.1 (from versions: 0.48.0)".
wheel==0.45.1
# execution-environment/requirements.txt's python-esiclient dependency is
# fetched from git (no pyproject.toml, legacy setup.py-only project), so
# pip builds its wheel using the default legacy build backend, whose
# implicit build-system.requires is ["setuptools", "wheel"] -- "wheel"
# itself is never otherwise pulled into the hermetic prefetch, confirmed
# live: "Could not find a version that satisfies the requirement wheel
# (from versions: none)".
wheel==0.48.0
# ansible-builder's introspect step merges every vendored collection's own
# bindep.txt/requirements.txt into the final combined pip requirements
# file at BUILD TIME (inside the Containerfile's builder stage, after
# collections are copied in) -- these entries never appear anywhere in
# osac-aap's own execution-environment/requirements.txt, so Cachi2 has no
# visibility into them from that file alone. Confirmed via a local
# `ansible-builder create` + introspect.py run against this component's
# actual collections/vendor tree (same merge the real build performs):
# pytz/python-dateutil/awxkit come from ansible.controller's own
# bindep.txt/requirements.txt; textfsm/ttp/xmltodict come from
# ansible.utils's -- all real, requested collections, not stale vendor
# (unlike the now-removed orphaned ansible.eda, which contributed a much
# bigger and genuinely unneeded set of these -- see its vendor removal
# commit).
pytz==2026.5
awxkit==24.6.1
textfsm==2.1.0
ttp==0.10.1
xmltodict==1.0.4
# cryptography/pydantic-core/rpds-py are maturin (Rust/PyO3) build-
# backend packages -- no pure-Python or prebuilt-wheel fallback, so
# building them from source hermetically (instead of fetching a wheel,
# which trips sbom_spdx.disallowed_package_attributes since it's
# sourced from public PyPI) needs maturin itself prefetched as a
# build-system requirement. Confirmed via each package's real sdist
# pyproject.toml: all three declare `build-backend = "maturin"`, none
# declare setuptools-rust. rust/cargo RPMs are prefetched in
# rpms.in.yaml/rpms.lock.yaml for the actual compilation.
maturin==1.15.0
# cryptography's own sdist build-system.requires also lists cffi
# (>=2.0.0 for python>=3.9, which this EE's python3.12 satisfies) --
# used by its own build.rs for the rust/cffi bridge, per its
# pyproject.toml's own comment.
cffi==2.1.1
# Both wheel pins above are themselves sdist-only in this fully-source
# build (no exception to the "no binary wheels" rule, including for

# flit_core>=3.11,<5; 3.12.0 satisfies both so one pin covers both.
# Confirmed live: "Could not find a version that satisfies the
# requirement flit_core<4,>=3.8 (from versions: none)". flit_core's own
# build-system.requires is empty (self-hosting), confirmed via its real
# sdist pyproject.toml, so no further transitive need here.
flit_core==3.12.0
# ansible-runner's build-system.requires also lists
# setuptools-scm[toml]>=6.2,<=8.1.0 (confirmed via its real sdist
# pyproject.toml) alongside setuptools -- 8.1.0 is the exact upper
# bound. Confirmed live: "Could not find a version that satisfies the
# requirement setuptools-scm<=8.1.0,>=6.2". setuptools-scm's own
# build-system.requires self-references (setuptools_scm>=8, a known
# bootstrap quirk of that package); its runtime requires_dist reduces

setuptools-scm==8.1.0
